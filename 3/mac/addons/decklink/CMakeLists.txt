cmake_minimum_required(VERSION 3.25)
project(vtr_pon_decklink_addon VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT APPLE OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "The Mac DeckLink add-on targets Apple Silicon arm64 only.")
endif()
option(VTRPON_DECKLINK_ENABLE_SDK "Build the real DeckLink media add-on" OFF)
set(VTRPON_DECKLINK_SDK_DIR "" CACHE PATH "Official DeckLink SDK 16.0 root containing Mac/include")
include(CTest)
add_library(decklink-media-core STATIC src/media_contract.cpp)
target_include_directories(decklink-media-core PUBLIC src)
set_target_properties(decklink-media-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(decklink-probe src/probe.cpp)
target_include_directories(decklink-probe PRIVATE include)
if(BUILD_TESTING)
  add_executable(decklink-media-test tests/media_test.cpp tests/c_header_test.c)
  target_include_directories(decklink-media-test PRIVATE include)
  target_link_libraries(decklink-media-test PRIVATE decklink-media-core)
  add_test(NAME decklink-media-contract COMMAND decklink-media-test)
endif()
if(VTRPON_DECKLINK_ENABLE_SDK)
  set(_sdk_include "${VTRPON_DECKLINK_SDK_DIR}/Mac/include")
  if(NOT EXISTS "${_sdk_include}/DeckLinkAPI.h" OR NOT EXISTS "${_sdk_include}/DeckLinkAPIDispatch.cpp")
    message(FATAL_ERROR "Set VTRPON_DECKLINK_SDK_DIR to the official SDK root containing Mac/include.")
  endif()
  file(STRINGS "${_sdk_include}/DeckLinkAPIVersion.h" _sdk_version REGEX "BLACKMAGIC_DECKLINK_API_VERSION_STRING")
  if(NOT _sdk_version MATCHES "16[.]0")
    message(FATAL_ERROR "This implementation targets DeckLink SDK 16.0; review other SDK versions explicitly.")
  endif()
  # Compile the official loader in place. SDK headers, dispatch source and the
  # Desktop Video driver/runtime are never copied into the repository/payload.
  add_library(vtr-pon-decklink-addon SHARED src/decklink_addon.cpp "${_sdk_include}/DeckLinkAPIDispatch.cpp")
  target_include_directories(vtr-pon-decklink-addon PRIVATE include "${_sdk_include}")
  target_compile_definitions(vtr-pon-decklink-addon PRIVATE VTRPON_DECKLINK_ADDON_BUILD=1)
  target_compile_options(vtr-pon-decklink-addon PRIVATE -Wall -Wextra)
  target_link_libraries(vtr-pon-decklink-addon PRIVATE decklink-media-core "-framework CoreFoundation")
  set_target_properties(vtr-pon-decklink-addon PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/decklink-addons")
  install(TARGETS vtr-pon-decklink-addon LIBRARY DESTINATION decklink-addons)
  if(BUILD_TESTING)
    add_executable(decklink-sdk-abi-test tests/sdk_abi_test.cpp)
    target_include_directories(decklink-sdk-abi-test PRIVATE include)
    target_link_libraries(decklink-sdk-abi-test PRIVATE vtr-pon-decklink-addon)
    add_test(NAME decklink-sdk-abi-no-hardware COMMAND decklink-sdk-abi-test)
  endif()
endif()
